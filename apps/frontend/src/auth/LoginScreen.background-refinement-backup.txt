const ForexCandleField: React.FC<{
  side: 'left' | 'right';
}> = ({ side }) => {
  const candles = side === 'left'
    ? FOREX_CANDLES.slice(0, 20)
    : FOREX_CANDLES.slice(20);

  const chartWidth = 330;
  const chartHeight = 440;
  const candleWidth = 13;
  const gap = 6;

  const minPrice = 55;
  const maxPrice = 350;

  const scaleY = (price: number) =>
    chartHeight -
    ((price - minPrice) / (maxPrice - minPrice)) * chartHeight;

  return (
    <svg
      className={`vs-forex-candles vs-forex-candles-${side}`}
      viewBox={`0 0 ${chartWidth} ${chartHeight}`}
      preserveAspectRatio="none"
      aria-hidden="true"
    >
      <defs>
        <filter
          id={`forexGlow-${side}`}
          x="-80%"
          y="-30%"
          width="260%"
          height="160%"
        >
          <feGaussianBlur stdDeviation="1.7" result="blur" />
          <feMerge>
            <feMergeNode in="blur" />
            <feMergeNode in="SourceGraphic" />
          </feMerge>
        </filter>
      </defs>

      {candles.map((candle, index) => {
        const x = 14 + index * (candleWidth + gap);

        const openY = scaleY(candle.open);
        const closeY = scaleY(candle.close);
        const highY = scaleY(candle.high);
        const lowY = scaleY(candle.low);

        const bullish = candle.close >= candle.open;

        const bodyTop = Math.min(openY, closeY);
        const bodyHeight = Math.max(
          Math.abs(closeY - openY),
          4
        );

        return (
          <g
            key={`${side}-forex-candle-${index}`}
            opacity={0.84 + (index % 5) * 0.035}
          >
            {/* upper/lower wick */}
            <line
              x1={x + candleWidth / 2}
              y1={highY}
              x2={x + candleWidth / 2}
              y2={lowY}
              stroke={bullish ? '#1BC5F8' : '#3A6382'}
              strokeWidth="1"
              filter={`url(#forexGlow-${side})`}
            />

            {/* candle body */}
            <rect
              x={x}
              y={bodyTop}
              width={candleWidth}
              height={bodyHeight}
              rx="1"
              fill={bullish ? '#078BC7' : '#12304A'}
              stroke={bullish ? '#28D0FF' : '#3C6685'}
              strokeWidth="0.7"
            />

            {/* subtle close/open edge */}
            <line
              x1={x}
              y1={bullish ? closeY : openY}
              x2={x + candleWidth}
              y2={bullish ? closeY : openY}
              stroke={bullish ? '#72E2FF' : '#557A96'}
              strokeWidth="0.65"
              opacity="0.72"
            />
          </g>
        );
      })}
    </svg>
  );
};

type WaveParticle = {
  x: number;
  y: number;
  radius: number;
  opacity: number;
  depth: number;
};

const WAVE_PARTICLES: WaveParticle[] = (() => {
  const particles: WaveParticle[] = [];

  const width = 900;
  const rows = 58;
  const columns = 96;

  for (let row = 0; row < rows; row++) {
    const vertical = row / (rows - 1);

    /*
     * Concentrate the mesh toward the lower portion
     * of the screen. Upper particles become progressively
     * sparse and faint.
     */
    const lowerBias = Math.pow(vertical, 2.15);

    for (let column = 0; column < columns; column++) {
      const horizontal = column / (columns - 1);

      const wave1 =
        Math.sin(horizontal * 12.2 + vertical * 5.4) * 24;

      const wave2 =
        Math.sin(horizontal * 19.5 - vertical * 8.2) * 11;

      const wave3 =
        Math.cos(horizontal * 7.4 + vertical * 13.1) * 8;

      const organicRipple =
        wave1 + wave2 + wave3;

      const baseY =
        610 +
        vertical * 350 +
        organicRipple;

      /*
       * Keep some upper particles but make them increasingly
       * sparse as the mesh approaches the black upper field.
       */
      const densityGate =
        vertical < 0.35
          ? ((column * 17 + row * 31) % 7 === 0)
          : true;

      if (!densityGate) continue;

      const edgeFade =
        Math.min(horizontal, 1 - horizontal) * 2;

      const opacity =
        (0.08 + lowerBias * 0.72) *
        Math.max(0.12, edgeFade);

      const radius =
        0.42 +
        lowerBias * 1.05 +
        (((column * 13 + row * 7) % 5) / 10);

      particles.push({
        x:
          horizontal * width +
          Math.sin(row * 1.71 + column * 0.19) * 3.2,

        y: baseY,

        radius,

        opacity,

        depth:
          0.28 +
          lowerBias * 0.72,
      });
    }
  }

  return particles;
})();

const ParticleWaveMesh: React.FC = () => {
  const connectionDistance = 42;

  return (
    <svg
      className="vs-wave-mesh"
      viewBox="0 0 900 1000"
      preserveAspectRatio="none"
      aria-hidden="true"
    >
      <defs>
        <radialGradient
          id="vs-wave-dot-gradient"
          cx="50%"
          cy="50%"
          r="50%"
        >
          <stop
            offset="0%"
            stopColor="#00E0FF"
            stopOpacity="1"
          />
          <stop
            offset="38%"
            stopColor="#00D4FF"
            stopOpacity="0.72"
          />
          <stop
            offset="100%"
            stopColor="#00B4FF"
            stopOpacity="0"
          />
        </radialGradient>

        <filter
          id="vs-wave-bloom"
          x="-200%"
          y="-200%"
          width="400%"
          height="400%"
        >
          <feGaussianBlur
            stdDeviation="2.2"
            result="blur"
          />
          <feMerge>
            <feMergeNode in="blur" />
            <feMergeNode in="SourceGraphic" />
          </feMerge>
        </filter>

        <linearGradient
          id="vs-wave-line"
          x1="0%"
          y1="0%"
          x2="100%"
          y2="0%"
        >
          <stop
            offset="0%"
            stopColor="#00B4FF"
            stopOpacity="0"
          />
          <stop
            offset="18%"
            stopColor="#00C8FF"
            stopOpacity="0.22"
          />
          <stop
            offset="50%"
            stopColor="#00E0FF"
            stopOpacity="0.38"
          />
          <stop
            offset="82%"
            stopColor="#00C8FF"
            stopOpacity="0.22"
          />
          <stop
            offset="100%"
            stopColor="#00B4FF"
            stopOpacity="0"
          />
        </linearGradient>
      </defs>

      {/* ------------------------------------------------------
          BACK / MID / FOREGROUND PARTICLE CONNECTION MESH
          ------------------------------------------------------ */}

      {WAVE_PARTICLES.map((particle, index) => {
        const nearby = WAVE_PARTICLES
          .slice(index + 1, index + 7)
          .map((candidate, candidateOffset) => ({
            candidate,
            candidateOffset,
          }))
          .filter(({ candidate }) => {
            const dx = candidate.x - particle.x;
            const dy = candidate.y - particle.y;
            return Math.sqrt(dx * dx + dy * dy) < connectionDistance;
          });

        return (
          <g key={`wave-network-${index}`}>
            {nearby.map(({ candidate, candidateOffset }) => (
              <line
                key={`wave-line-${index}-${candidateOffset}`}
                x1={particle.x}
                y1={particle.y}
                x2={candidate.x}
                y2={candidate.y}
                stroke="url(#vs-wave-line)"
                strokeWidth={
                  particle.depth > 0.72 ? 0.7 : 0.42
                }
                opacity={
                  particle.opacity *
                  candidate.opacity *
                  2.2
                }
              />
            ))}

            <circle
              cx={particle.x}
              cy={particle.y}
              r={particle.radius * 2.7}
              fill="url(#vs-wave-dot-gradient)"
              opacity={particle.opacity * 0.24}
              filter="url(#vs-wave-bloom)"
            />

            <circle
              cx={particle.x}
              cy={particle.y}
              r={particle.radius}
              fill="#00D4FF"
              opacity={particle.opacity}
            />
          </g>
        );
      })}
    </svg>
  );
};
